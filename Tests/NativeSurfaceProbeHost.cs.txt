using System;
using System.Management.Automation;
using System.Management.Automation.Runspaces;
using System.Threading;

internal static class NativeSurfaceProbeHost
{
    [STAThread]
    private static int Main(string[] args)
    {
        if (args.Length != 1) return 2;
        try
        {
            using (Runspace runspace = RunspaceFactory.CreateRunspace())
            {
                runspace.ApartmentState = ApartmentState.STA;
                runspace.ThreadOptions = PSThreadOptions.UseCurrentThread;
                runspace.Open();
                using (PowerShell shell = PowerShell.Create())
                {
                    shell.Runspace = runspace;
                    shell.AddCommand(args[0]);
                    foreach (PSObject item in shell.Invoke()) Console.WriteLine(item);
                    foreach (ErrorRecord error in shell.Streams.Error) Console.Error.WriteLine(error);
                    return shell.HadErrors ? 1 : 0;
                }
            }
        }
        catch (Exception ex)
        {
            Console.Error.WriteLine(ex);
            return 1;
        }
    }
}
